测试推送 (使用 curl)

# 推送给指定用户
curl -X POST http://localhost:9090/api/push \
  -H "Content-Type: application/json" \
  -d '{
    "user_ids": ["user001", "user002"],
    "payload": {
      "title": "测试消息",
      "content": "这是一条推送消息",
      "time": "'$(date +%Y-%m-%d\ %H:%M:%S)'"
    }
  }'

# 广播给所有用户
curl -X POST http://localhost:9090/api/broadcast \
  -H "Content-Type: application/json" \
  -d '{
    "payload": {
      "type": "broadcast",
      "content": "这是一条广播消息"
    }
  }'
